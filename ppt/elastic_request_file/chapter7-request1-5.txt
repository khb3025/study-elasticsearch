GET /boards/_search
// 광고 게시판의 글이 아니면서
// 공지글이 아니면서 좋겠다.
// '검색엔진'이랑 관련된 게시글을 찾고 싶다.

GET /boards/_search
{
    "query": {
        "bool": {
            "must_not": [
                {
                    "term": {
                        "category": "광고 게시판"
                    }
                    
                },
                {
                    "term": {
                      "is_active": {
                        "value": true
                      }
                    }
                }
            ],
            "must": [
                {
                    "match": {
                      "title": "검색엔진"
                    }
                }
            ]
        }
    }
}
