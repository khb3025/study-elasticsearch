        }
    },
    "mappings": {
        "properties": {
             "name": {
                "type": "text",

            },
            "likes": {
                "type": "integer"
            }
        }
    }
}

GET /products/_search
// 사용자가 특정 키워드 (예: "무선 이어폰")으로 검색했을때,
// 키워드와 관련된 데이터 조회
// 평점이 높거나 
// 좋아요 수가 많은 상품 상단 노출
GET /products/_search
{
    "query": {
        
        "bool": {
            "must": [
                { 
                    "match": {
                    "name": "무선 이어폰"
                    }
                }
            ],
            "should": [
                {
                    "range": {
                        "rating" : {
                            "gte": 4.5
                        }
                    }
                },
                {
                    "range": {
                        "likes": {
                            "gte": 100
                        }
                    }
                }
            ]
        }
    }
}
